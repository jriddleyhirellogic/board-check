116static void v_Init_Register_Entry( uint8_t u8_ID,
117 uint32_t u32_Reset_Value,
118 Read_Callback_t read_fn,
119 Write_Callback_t write_fn )
120{
121 assert( u8_ID < NUMBER_REGISTERS );
122 Register_Entry_t *p_Reg = &Register_Table[u8_ID];
123 p_Reg->u32_Data = u32_Reset_Value;
124 p_Reg->read = read_fn;
125 p_Reg->write = write_fn;
126}

135Register_Entry_t *Get_Register_Entry( uint16_t u16_ID )
136{
137 Register_Entry_t *p_Entry = (Register_Entry_t *)0;
138 if( u16_ID < NUMBER_REGISTERS && Register_Table[u16_ID].read != REG_UNUSED )
139 p_Entry = &Register_Table[u16_ID];
140 return p_Entry;
141}

151Error_Code_t Read_Register( uint16_t u16_ID, uint32_t *read_buffer )
152{
153 Error_Code_t retval = NO_ERROR;
154 Register_Entry_t *p_Reg = Get_Register_Entry( u16_ID );
155
156 if( p_Reg == (Register_Entry_t *)0 )
157 {
158 retval = ERR_REG_INVALID_ADDR;
159 }
160 else
161 {
162 if(p_Reg->read) // Callback overrides data element
163 {
164 retval = p_Reg->read(read_buffer);
165 }
166 else // No callback
167 {
168 *read_buffer = p_Reg->u32_Data;
169 retval = NO_ERROR;
170 }
171 }
172
173 return retval;
174}

184Error_Code_t Write_Register( uint16_t u16_ID, uint32_t u32_write_data )
185{
186 Error_Code_t retval = NO_ERROR;
187 Register_Entry_t *p_Reg = Get_Register_Entry( u16_ID );
188
189 if( p_Reg == (Register_Entry_t *)0 )
190 {
191 retval = ERR_REG_INVALID_ADDR;
192 }
193 else
194 {
195 if(p_Reg->write) // Callback overrides
196 {
197 retval = p_Reg->write( u32_write_data );
198 }
199 else
200 {
201 p_Reg->u32_Data = u32_write_data;
202 retval = NO_ERROR;
203 }
204 }
205
206 return retval;
