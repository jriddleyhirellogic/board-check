45 v_Set_LVDT_Gain( LVDT_GAIN_8 );
46 v_Set_Microstep_Mode( STEP_256 );

50 case EXIT:
51 v_Stop_Focus_Timer();
52 v_Set_Stepper_out( 0 );
53 break;
54
55 case FOCUS_TIMER_EXPIRED:
56 v_Start_Focus_Timer(10);
57 //v_Set_Debug_GPIO( J7_PIN_6 , J7_PIN_6 );
58
59 V_Get_LVDT_Snapshot( &LVDT_Sample );
60 i32_Projection = i32_Get_Projection( &LVDT_Sample );
61
62 i32_Direction = i32_Projection > 0 ? 1 : -1;
63
64 v_Set_Stepper_out( -i32_Direction * STEP_SPEED ); // DISP < 0 V < 0 : DISP > 0 V > 0
65
66 if( i32_Direction != i32_Original_Direction )
67 next_state = (State_t)Idle_State;
68
69 break;
70
71 default:
72 break;
73 }
