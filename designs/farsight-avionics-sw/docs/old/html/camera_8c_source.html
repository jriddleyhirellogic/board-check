158void configure_udp(uint32_t base_addr,
159 uint16_t dst_port,
160 uint16_t src_port,
161 uint32_t dst_ip,
162 uint32_t src_ip,
163 uint16_t dst_mac_msb,
164 uint32_t dst_mac_lsb,
165 uint16_t src_mac_msb,
166 uint32_t src_mac_lsb)
167{
168 uint32_t reg32_status = 0;
169
170 // Configure UDP ports
171 HAL_set_32bit_reg(base_addr, UDP_DST_PORT, dst_port);
172 reg32_status = HAL_get_32bit_reg(base_addr, UDP_DST_PORT);
173
174 HAL_set_32bit_reg(base_addr, UDP_SRC_PORT, src_port);
175 reg32_status = HAL_get_32bit_reg(base_addr, UDP_SRC_PORT);
176
177 // Configure IP addresses
178 HAL_set_32bit_reg(base_addr, DST_IP, dst_ip);
179 reg32_status = HAL_get_32bit_reg(base_addr, DST_IP);
180
181 HAL_set_32bit_reg(base_addr, SRC_IP, src_ip);
182 reg32_status = HAL_get_32bit_reg(base_addr, SRC_IP);
183
184 // Configure destination MAC address
185 HAL_set_32bit_reg(base_addr, DST_MAC_MSB, dst_mac_msb);
186 reg32_status = HAL_get_32bit_reg(base_addr, DST_MAC_MSB);
187
188 HAL_set_32bit_reg(base_addr, DST_MAC_LSB, dst_mac_lsb);
189 reg32_status = HAL_get_32bit_reg(base_addr, DST_MAC_LSB);
190
191 // Configure source MAC address
192 HAL_set_32bit_reg(base_addr, SRC_MAC_MSB, src_mac_msb);
193 reg32_status = HAL_get_32bit_reg(base_addr, SRC_MAC_MSB);
194
195 HAL_set_32bit_reg(base_addr, SRC_MAC_LSB, src_mac_lsb);
196 reg32_status = HAL_get_32bit_reg(base_addr, SRC_MAC_LSB);
197
198 (void)reg32_status;
199}
