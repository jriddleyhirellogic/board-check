21void v_Update_Camera_State( Camera_Context_t *p_context, Event_t *p_event )
22{
23 // Handle immediate events - doesn't matter what state you are in
24 State_t next_state = 0;
25 State_t immediate = (State_t)0;
26
27 // events that must be handled regardless of current state

33
34 case CAMERA_MODE_SET:

36 {
37 case CAMERA_LP_STATE:
38 immediate = (State_t)Camera_Low_Power_State;
39 break;
40
41 case CAMERA_IDLE_STATE:

43 break;
44

47 break;
48

50 immediate = (State_t)Camera_Busy_State;
51 break;
52
53 case CAMERA_FAULT_STATE:
54 immediate = (State_t)Camera_Fault_State;
55
56 default:
57 break;
58 }
59 break;
60
61 case RESET_RECEIVED:
62 immediate = (State_t)Camera_Low_Power_State;
63 break;
64 default:
65 break;
66 }
67
68
69 if( immediate == (State_t)0 )
70 {
71 next_state = ((Camera_State_Handler_t)p_context->current_state)( p_context, p_event );
72 }
73 else
74 {
75 next_state = immediate;
76 }
77
78 if( next_state != p_context->current_state )
79 {
80 Event_t event;
81 event.no_data.sig = EXIT;
82
83 ((Camera_State_Handler_t)p_context->current_state)( p_context, &event );
84
85 p_event->no_data.sig = ENTRANCE;
86 ((Camera_State_Handler_t)next_state)( p_context, p_event );
87
88 p_context->current_state = next_state;
89
90 }
