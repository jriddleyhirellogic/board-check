47void v_Serial_Comm_Task(void *pvParameters)
48{
49 Serial_Comm_Queue = xQueueCreate( SERIAL_COMM_QUEUE_LENGTH,
50 sizeof(Event_t) );
51
52 Reset_Serial_SM();
53
54 xEventGroupSetBits( initialized, SERIAL_COMM_INITIALIZED );
55 xEventGroupWaitBits( initialized, ALL_INITIALIZED, pdFALSE, pdTRUE, portMAX_DELAY );
56
57 v_Enable_Serial_Comm();
58
59 while(1)
60 {
61 BaseType_t b_Data_Present;
62 Event_t Rcvd_Event;
63 b_Data_Present = xQueueReceive(
64 Serial_Comm_Queue,
65 &Rcvd_Event,
66 portMAX_DELAY );
67
68 if( b_Data_Present ) // Event received
69 {
70 uint8_t u8_Response_Length = 0;
71 uint8_t u8_Retval;
72
73 // Parse the received packet
74 u8_Retval = Execute_Command( &Rcvd_Event.serial_msg.header,
75 Rcvd_Event.serial_msg.Data,
76 u8_Response,
77 &u8_Response_Length );
78
79 if( !u8_Retval )
80 v_Send_Serial_Comm( u8_Response, u8_Response_Length );
81
82 ++u8_Msgs_Received;
83 }
84 else // timeout or something wrong
85 {

134void v_UART16550_Handler(void)
135{
136 uint8_t u8_Line_Status = u8_Get_Line_Status( p_serial_comm_instance );
137 uint8_t b_Rx_Data_Present = (u8_Line_Status & LINE_STATUS_DR) != 0;
138 uint8_t b_Tx_Holding_Reg_Empty = (u8_Line_Status & LINE_STATUS_THRE) != 0;
139 uint8_t u8_IRQ_Cause = u8_Get_Interrupt_Cause( p_serial_comm_instance );
140
141
142 if( (u8_IRQ_Cause & IRQ_ID_RX_DATA_AVAIL ) != 0 && b_Rx_Data_Present )
143 {
144 uint8_t u8_Data = u8_Get_UART_Rx_Data( p_serial_comm_instance );
145 v_Update_Serial_SM(u8_Data);
146
147 // The reload value corresponds to 3 character times at 115200 bits/sec
148 Reload_Comm_Timer( 6510 );
149 ++u8_Rx_Count;
150 }
151
152 // Transmit
153 if( (u8_IRQ_Cause & IRQ_TX_HOLD_REG_EMPTY ) != 0 )
154 {
155 if( u8_Tx_Count_Remaining != 0 )
156 {
157 v_Write_UART_Tx_Data( p_serial_comm_instance, *p_Tx_Ptr++ );
158 --u8_Tx_Count_Remaining;
159 }
160 else
161 {
162 v_Disable_UART_IRQ( p_serial_comm_instance, UART_ETBEI );
163 }
164 }
