49void v_Camera_Task(void *pvParameters)
50{
51 static uint32_t u32_Counter;
52 uint8_t u8_Timer_ID;
53
54 Camera_Context.instance = &camera_instance;
55
56 // Camera initialization is being done here because there is no systick in early boot.
57 v_Init_Camera( Camera_Context.instance );
58
59 // Queue for events directed at control thread
60 Camera_Queue = xQueueCreate( CAMERA_QUEUE_LENGTH , sizeof(Event_t) );
61
62 // Camera timer
63 Camera_Timer = xTimerCreate(
64 "Camera_Timer",
65 100, // Default period (mS)
66 pdFALSE, // One shot, no auto-reload
67 &u8_Timer_ID,
68 Camera_Timer_Expiration );
69
70 v_Init_Camera_State_Machine( &Camera_Context, Camera_Low_Power_State );
71
72 // Initialization complete
73 xEventGroupSetBits( initialized, CAMERA_INITIALIZED );
74
75 // Wait for all tasks to be initialized
76 xEventGroupWaitBits( initialized, ALL_INITIALIZED, pdFALSE, pdTRUE, portMAX_DELAY );
77
78 while(1)
79 {
80 BaseType_t b_Event_Received;
81 Event_t Rcvd_Event;
82
83 b_Event_Received = xQueueReceive( Camera_Queue,
84 &Rcvd_Event,
85 portMAX_DELAY );
86
87 if( b_Event_Received) // Event received
88 {
89 v_Update_Camera_State( &Camera_Context, &Rcvd_Event );
90 }
91 else // Timeout for period updates
92 {
93 }
94 }
95}
