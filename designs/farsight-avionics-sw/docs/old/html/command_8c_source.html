54uint8_t Execute_Command( Serial_Msg_Header_t *p_Header,
55 uint8_t *p_u8_Data,
56 uint8_t *p_Response_Buffer,
57 uint8_t *p_u8_Response_Length )
58{
59 uint8_t u8_Arg;
60 uint32_t u32_Arg;
61 uint32_t u32_Test_Val;
62 uint8_t u8_Ret_Val = OK;
63 uint8_t u8_Register_Error_Code;
64 uint8_t u8_Error_Code = 0;
65 LVDT_Sample_t LVDT_Sample;
66 int32_t i32_Projection,
67 i32_Displacement;
68 Event_t event;
69 Status_t Status;
70 uint8_t u8_Reg_ID;
71
72 v_Init_Response( p_Header, p_Response_Buffer );
73
74 switch( p_Header->u8_Opcode )
75 {
76 case GET_VERSION_SYS_CMD:
77 u32_Arg = SYS_VERSION;
78 v_Add_Response_Data( &u32_Arg, sizeof(&u32_Arg), p_Response_Buffer );
79 break;
80 case SET_MODE_SYS_CMD: // Forces changes to camera mode
81 u32_Arg = *p_u8_Data;
82 switch( u32_Arg )
83 {
84 case 0: // LP Mode
85 event.no_data.sig = POWER_DOWN_CAMERA;
86 v_Post_Camera_Event( &event );
87 break;
88
89 case 1: // Update Mode Not implemented yet
90 break;
91
92 case 2: // Idle Mode
93 event.no_data.sig = POWER_UP_CAMERA;
94 v_Post_Camera_Event( &event );
95 break;
96
97 case 3: // Manual Mode
98 event.no_data.sig = ARM_CAMERA;
99 v_Post_Camera_Event( &event );
100 break;
101
102 case 4: // Stream Mode Not implemented yet
103 break;
104
105 default:
106 break;
107 }
108 break;
109 case RESET_SYS_CMD:
110 // System reset
111 event.no_data.sig = RESET_RECEIVED;
112 v_Post_System_Event( &event );
113
114 // Camera reset
115 event.no_data.sig = RESET_RECEIVED;
116 v_Post_Camera_Event( &event );
117 break;
118
119 case REG_READ_SYS_CMD:
120 u8_Reg_ID = *p_u8_Data;
121 u8_Register_Error_Code = Read_Register( u8_Reg_ID, &u32_Test_Val );
122 v_Set_Error_Code( u8_Register_Error_Code, p_Response_Buffer );
123 v_Add_Response_Data( &u32_Test_Val, sizeof(&u32_Test_Val), p_Response_Buffer );
124 break;
125
126 case REG_WRITE_SYS_CMD:
127 u8_Reg_ID = *p_u8_Data;
128 memcpy(&u32_Arg, p_u8_Data+1, sizeof(u32_Arg)); // Argument may be incorrectly aligned
129 u8_Register_Error_Code = Write_Register( u8_Reg_ID, u32_Arg );
130 v_Set_Error_Code( u8_Register_Error_Code, p_Response_Buffer );
131 break;
132
133 case SET_TIME_SYS_CMD:
134 break;
135
136 case START_ACQ_SYS_CMD:
137 event.no_data.sig = CAMERA_TAKE_PICTURE;
138 v_Post_Camera_Event( &event );
139 break;
140
141 case STOP_ACQ_SYS_CMD:
142 break;
143
144 case FOCUS_SYS_CMD:
145 u8_Error_Code = u8_Calculate_Focus_Distance_microns( u32_Arg, &event.position.i32_Position );
146 memcpy(&u32_Arg, p_u8_Data, sizeof(uint32_t ));
147 event.position.sig = GOTO_POSITION;
148 v_Post_Focus_Event( &event );
149 v_Set_Error_Code( u8_Error_Code, p_Response_Buffer );
150 break;
151
152 case HALT_FOCUS_SYS_CMD:
153 break;
154
155 case XFER_BUFF_SYS_CMD:
156 break;
157
158 case RESET_BUFF_SYS_CMD:
159 break;
160
161 case GET_INFO_SYS_CMD:
162 break;
163
164 case CONF_EVENT_LOG_SYS_CMD:
165 break;
166
167 case EN_TL_CHAN_LOG_SYS_CMD:
168 break;
169
170 case XFER_LOG_SYS_CMD:
171 break;
172
173 // User defined are below for test and integration
174 case ECHO_FUNCTION:
175 // This should simply reflect back the received packet for testing
176 v_Add_Response_Data( p_u8_Data, p_Header->u8_Data_Length, p_Response_Buffer );
177 break;
178
179 case SET_POSITION: // Target position in microns
180 memcpy(&u32_Arg, p_u8_Data, sizeof(uint32_t ));
181 event.position.sig = GOTO_POSITION;
182 event.position.i32_Position = (int32_t)u32_Arg;
183 v_Post_Focus_Event( &event );
184 v_Set_Error_Code(0, p_Response_Buffer );
185 break;
186
187 case ZERO_FUNCTION:
188 event.no_data.sig = GOTO_POSITION;
189 event.position.i32_Position = 0;
190 v_Post_Focus_Event( &event );
191 break;
192
193 case SET_LVDT_OFFSET_FUNCTION: // sets offset in microns
194 memcpy(&u32_Arg, p_u8_Data, sizeof(uint32_t ));
195 v_Set_Displacement_Offset( (int32_t) u32_Arg );
196 v_Set_Error_Code(0, p_Response_Buffer );
197 break;
198
199 case GET_POSITION_FUNCTION:
200 i32_Displacement = i32_Get_Position();
201 v_Add_Response_Data( &i32_Displacement, sizeof(i32_Displacement), p_Response_Buffer );
202 break;
203
204 case GET_STATUS_FUNCTION:
205 v_Get_Status( &Status );
206 v_Add_Response_Data( &Status.i32_Position_Microns, sizeof( Status.i32_Position_Microns), p_Response_Buffer );
207 v_Add_Response_Data( &Status.i32_Velocity, sizeof( Status.i32_Velocity ), p_Response_Buffer );
208 v_Add_Response_Data( &Status.u8_LVDT_Gain, sizeof( Status.u8_LVDT_Gain ), p_Response_Buffer );
209 break;
210
211 case RESET_FUNCTION:
212 event.no_data.sig = RESET_RECEIVED;
213 v_Post_Focus_Event( &event );
214 break;
215
216 case GET_PROJECTION: // Mainly for debugging
217 V_Get_LVDT_Snapshot( &LVDT_Sample );
218 i32_Projection = i32_Get_Projection( &LVDT_Sample );
219 v_Add_Response_Data( &i32_Projection, sizeof( i32_Projection ), p_Response_Buffer );
220 break;
221
222
223 case SET_VREF_FUNCTION:
224 case SET_CONTROL_OUTPUTS_FUNCTION:
225 default: // well formed but not recognized
226 v_Set_Error_Code( ERR_CMD_INVALID_OPCODE, p_Response_Buffer );
227 break;
228 }
229
230 v_Finalize_Msg( p_Response_Buffer, p_u8_Response_Length );
231 return u8_Ret_Val;

248static void v_Init_Response( Serial_Msg_Header_t *p_Header, uint8_t *p_Response_Buffer )
249{
250 u8_Response_Length = 0;
251 // Sync field
252 p_Response_Buffer[u8_Response_Length++] = SYNC1_CODE;
253 p_Response_Buffer[u8_Response_Length++] = SYNC2_CODE;
254 p_Response_Buffer[u8_Response_Length++] = SYNC3_CODE;
255 p_Response_Buffer[u8_Response_Length++] = SYNC4_CODE;
256
257 // Header
258 p_Response_Buffer[u8_Response_Length++] = p_Header->u8_Opcode;
259 p_Response_Buffer[u8_Response_Length++] = p_Header->u8_Error_Code;
260 p_Response_Buffer[u8_Response_Length++] = p_Header->u8_Sequence;
261 p_Response_Buffer[u8_Response_Length++] = 0; // Data length before anything is added
262
263 memset( p_Response_Buffer + u8_Response_Length, 0, RESERVED_SIZE ); // Rserved field
264 u8_Response_Length += RESERVED_SIZE;
265}

293static void v_Finalize_Msg( uint8_t *p_Response_Buffer, uint8_t *p_u8_Response_Length )
294{
295 // Set real data size
296 p_Response_Buffer[DATA_LENGTH_OFFSET] = u8_Response_Length - (SYNC_LENGTH + HEADER_SIZE);
297 // Calculate over all fields except for sync
298 uint32_t u32_CRC = crc( p_Response_Buffer + 4, u8_Response_Length - 4, 0 );
299 p_Response_Buffer[u8_Response_Length++]= (u32_CRC >> 24) & 0xFF;
300 p_Response_Buffer[u8_Response_Length++]= (u32_CRC >> 16) & 0xFF;
301 p_Response_Buffer[u8_Response_Length++]= (u32_CRC >> 8) & 0xFF;
302 p_Response_Buffer[u8_Response_Length++]= u32_CRC & 0xFF;
303
304 *p_u8_Response_Length = u8_Response_Length;
305}
