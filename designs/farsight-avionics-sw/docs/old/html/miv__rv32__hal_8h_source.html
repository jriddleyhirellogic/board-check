461static inline void MRV_raise_soft_irq(void)
462{
463 set_csr(mie, MIP_MSIP); /* Enable software interrupt bit */
464
465#ifdef MIV_LEGACY_RV32

467 MSIP = 0x01; /* raise soft interrupt */

469 /* Raise soft IRQ on MIV_RV32 processor */

471#endif
472}

507/***************************************************************************/
510static inline uint64_t MRV_read_mtime(void)
511{
512 volatile uint32_t hi = 0u;
513 volatile uint32_t lo = 0u;
514
515 /* when mtime lower word is 0xFFFFFFFF, there will be rollover and
516 * returned value could be wrong. */
517 do {
518 hi = MTIMEH;
519 lo = MTIME;
520 } while(hi != MTIMEH);
521
522 return((((uint64_t)MTIMEH) << 32u) | lo);
523}
