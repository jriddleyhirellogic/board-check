42void v_Focus_Task(void *pvParameters)
43{
44 uint8_t u8_Timer_ID;
45 TickType_t Control_Update_Period = pdMS_TO_TICKS(10); // For periodic tasks
46
47 // Queue for events directed at control thread
48 Control_Queue = xQueueCreate( CONTROL_QUEUE_LENGTH , sizeof(Event_t) );
49
50 // Control timer
51 Focus_Timer = xTimerCreate(
52 "Focus_Timer",
53 100, // Default period (mS)
54 pdFALSE, // One shot, no auto-reload
55 &u8_Timer_ID,
56 Focus_Timer_Expiration );
57
58 Focus_Fault_Timer = xTimerCreate(
59 "Control_Fault",
60 100, // Default period (mS)
61 pdFALSE, // One shot, no auto-reload
62 &u8_Timer_ID,
63 Focus_Fault_Timer_Expiration );
64
65
66 // Initialization complete
67 xEventGroupSetBits( initialized, CONTROL_INITIALIZED );
68
69 // Wait for all tasks to be initialized
70 xEventGroupWaitBits( initialized, ALL_INITIALIZED, pdFALSE, pdTRUE, portMAX_DELAY );
71
72 v_Stop_Focus_Timer();
73 focus_context.p_PID_Instance = &PID_Instance;
74
75 v_Init_Focus_State_Machine( &focus_context, Initial_State );
76
77 while(1)
78 {
79 BaseType_t b_Event_Received;
80 Event_t Rcvd_Event;
81
82 b_Event_Received = xQueueReceive( Control_Queue,
83 &Rcvd_Event,
84 portMAX_DELAY );
85
86 if( b_Event_Received) // Event received
87 {
88 v_Update_Focus_State( &focus_context, &Rcvd_Event );
89 }
90 else // Timeout for period updates
91 {
92 ++u32_Timeout_Counter;
93 }
94
95#ifdef MEM_CHECK
96 // Measure remaining stack space
97 UBaseType_t uxHighWaterMark = uxTaskGetStackHighWaterMark(NULL);
98#endif
99
100 }
101}
