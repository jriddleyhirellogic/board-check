56/*
57 * The relationship between the data duty cycle and the current setpoint is:
58 * I_FS = (V_Ref / K_V) * DC (%)
59 *
60 * => DC(%) = (I_FS(mA) * 1.32) / 3200 (mV)
61 */

70void PWM_Set_Voltage_mV( PWM_Instance_t *instance, uint16_t u16_V_Set_mV )
71{
72 uint32_t u32_Temp;
73 uint16_t u16_DC_Setpoint;
77 u32_Temp = COUNT_PER_MV * (uint32_t)u16_V_Set_mV;
78 u16_DC_Setpoint = (uint16_t)(u32_Temp >> 10 );
79 HAL_set_16bit_reg( instance->u32_Base_Address, PWM1_NEGEDGE, u16_DC_Setpoint );
80}
