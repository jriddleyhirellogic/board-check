137int32_t i32_Get_Projection( LVDT_Sample_t *p_LVDT_Sample )
138{
139 // This calculates the projection of the secondary response onto the primary viewing them as 2D vectors
140 // formed by (I,Q)
141 // The projection is scaled to increase precision.
142
143 int64_t numerator = (int64_t)p_LVDT_Sample->i32_I_Primary * (int64_t)p_LVDT_Sample->i32_I_Secondary +
144 (int64_t)p_LVDT_Sample->i32_Q_Primary * (int64_t)p_LVDT_Sample->i32_Q_Secondary;
145
146 int64_t denom = ((int64_t)p_LVDT_Sample->i32_I_Primary * (int64_t)p_LVDT_Sample->i32_I_Primary ) +
147 ((int64_t)p_LVDT_Sample->i32_Q_Primary * (int64_t)p_LVDT_Sample->i32_Q_Primary );
148
149 // Calculates the ratio of the length of the projection of S along P to the norm of P
150 // r = (S dot P)/ norm(P)^2
151
152 int64_t i64_Projection = (numerator * LVDT_SCALING_FACTOR ) / denom;
153
154 return (int32_t)i64_Projection;
155}

165float f_Get_Displacement( int32_t i32_Projection )
166{
167 Cal_Value_t *table;
168 uint8_t u8_Table_Size;
169 LVDT_Gain_Sel_t e_Gain_Sel;
170
172 e_Gain_Sel = e_Get_LVDT_Gain();
173
174 switch( e_Gain_Sel )
175 {
176 case LVDT_GAIN_1:
177
178 u8_Table_Size = sizeof(Cal_G1)/sizeof(Cal_Value_t);
179 table = (Cal_Value_t *)Cal_G1;
180 break;
181
182 case LVDT_GAIN_8:
183 u8_Table_Size = sizeof(Cal_G8)/sizeof(Cal_Value_t);
184 table = Cal_G8;
185 break;
186
187 default:
188 break;
189 }
190
191 uint8_t u8_n;
192 float f_Displacement;
193
195 if( i32_Projection < table[u8_Table_Size-1].i32_Projection )
196 return -999999.0f;
197
198 if( i32_Projection > table[0].i32_Projection )
199 return 999999.0f;
200
202 for( u8_n = 0; u8_n < (u8_Table_Size - 1); u8_n++ )
203 {
204 if( i32_Projection > table[u8_n+1].i32_Projection )
205 break;
206 }
207
208 f_Displacement = f_Interpolate_2D( (float) i32_Projection,
209 (float)table[u8_n].i32_Projection,
210 (float)table[u8_n+1].i32_Projection,
211 table[u8_n].f_Displacement,
212 table[u8_n+1].f_Displacement );
213
214
215
216 return f_Displacement;
217}

229static float f_Interpolate_2D( float x, float x0, float x1, float y0, float y1 )
230{
231 float y = (x - x0) * y1 + (x1 - x) * y0;
232 y /= (x1 - x0 );
233
234 return y;
235}
