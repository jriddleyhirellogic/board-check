140static void v_Set_UART_Divisor( UART_16550_Instance_t *this_instance, uint16_t u16_Divisor )
141{
142 uint8_t u8_Temp;
143 uint8_t u8_Readback;
144
145 // @todo enter critical section
146 // Set Data latch access bit
147 u8_Temp = HAL_get_8bit_reg( this_instance->u32_Base_Address, LINE_CTRL );
148 u8_Temp |= UART16550_DLAB;
149 HAL_set_8bit_reg(this_instance->u32_Base_Address, LINE_CTRL, u8_Temp);
150
151 // Set MSB
152 HAL_set_8bit_reg(this_instance->u32_Base_Address, DIVISOR_LATCH_MSB, u16_Divisor >> 8);
153 u8_Readback = HAL_get_8bit_reg( this_instance->u32_Base_Address, DIVISOR_LATCH_MSB );
154
155 // Set LSB
156 HAL_set_8bit_reg(this_instance->u32_Base_Address, DIVISOR_LATCH_LSB, u16_Divisor & 0xFF);
157 u8_Readback = HAL_get_8bit_reg( this_instance->u32_Base_Address, DIVISOR_LATCH_LSB );
158
159 // Clear Data latch access bit
160 u8_Temp &= ~UART16550_DLAB;
161 HAL_set_8bit_reg(this_instance->u32_Base_Address, LINE_CTRL, u8_Temp );
162
163 // @todo exit critical section
164}
