55int32_t i32_Update_PID( PID_Instance_t *instance, int32_t i32_Setpoint, int32_t i32_Feedback,int32_t i32_Feedforward )
56{
57 int32_t i32_Retval = 0;
58 int32_t i32_Proportional_Term = 0;
59 int32_t i32_integral_term = 0;
60 int32_t i32_Non_Integral_Sum = 0;
61 int state;
62 int32_t i32_Save;
63 float f_Error = 0.0f;
64 int8_t before, after;
65
66 int32_t i32_Error = i32_Setpoint - i32_Feedback;
67
68 // Use only feedforward if error is within deadband
69#ifdef USE_FEEDFORWARD_W_DEADBAND
70 if( ABS(i32_Error) <= instance->deadband )
71 {
72 instance->i32_Integral_Sum = 0;
73 i32_Retval = i32_Feedforward;
74 state = 0;
75 }
76#endif
77
78#ifdef USE_INTEGRAL_DEADBAND
79 if( ABS(i32_Error) <= instance->integral_deadband )
80 {
81 // Use only proportional part if error is withn integral deadband
82 instance->i32_Integral_Sum = 0;
83 i32_Proportional_Term = (int32_t)((float)i32_Error * instance->k_p);
84 i32_Retval = i32_Clamp( i32_Proportional_Term + i32_Feedforward,
85 instance->min_out,
86 instance->max_out );
87 state = 1;
88 }
89#endif
90
91 f_Error = (float)i32_Error;
92
93 i32_Proportional_Term = (int32_t)( f_Error * instance->k_p );
94
95 // Proportional part
96 i32_Non_Integral_Sum = i32_Feedforward + i32_Proportional_Term;
97
98 // Integral part
99 i32_integral_term = (int32_t)( f_Error * instance->k_i );
100
101 instance->i32_Integral_Sum += i32_integral_term;
102
103 // Bounds the integral portion
104 instance->i32_Integral_Sum = i32_Clamp( instance->i32_Integral_Sum,
105 instance->min_out - i32_Non_Integral_Sum,
106 instance->max_out - i32_Non_Integral_Sum );
107
108 i32_Retval = ( i32_Non_Integral_Sum + instance->i32_Integral_Sum );
109 state = 2;
110
111
112 taskENTER_CRITICAL();
113 Log_Info.i32_Position = i32_Feedback;
114 Log_Info.i32_Setpoint = i32_Setpoint;
115 Log_Info.i32_Error = i32_Error;
116 Log_Info.i32_Integral_Sum = instance->i32_Integral_Sum;
117 Log_Info.i32_Output = i32_Retval;
118 Log_Info.state = state;
119 taskEXIT_CRITICAL();
120
121
122 return i32_Retval;
123}
