63//------------------------------------------------------------------------------
64// Eth PHY setup functions
65//------------------------------------------------------------------------------
66void phy_advertise(eth_core_regs_t* regs)
67{
68 uint32_t phy_reg = 0xFFFF;
69
70 // Device Auto-Negotiation Advertisement, Address 4 (0x04)
71 // The bits in address 4 in the main registers space control the ability to
72 // notify other devices of the status of
73 // its auto-negotiation feature.
74 *(regs->mdio_mgmt_address_reg) = 0x0004;
75 *(regs->mdio_mgmt_control_reg) = 0x1;
76 while (*(regs->mdio_mgmt_indicators_reg) != 0)
77 ;
78 *(regs->mdio_mgmt_command_reg) = 0x1;
79 while (*(regs->mdio_mgmt_indicators_reg) != 0)
80 ;
81 phy_reg = *(regs->mdio_mgmt_status_reg);
82 *(regs->mdio_mgmt_command_reg) = 0x0;
83
84 phy_reg &= ~(0x1E); // 0b11100001
85
86 *(regs->mdio_mgmt_address_reg) = 0x0004;
87 *(regs->mdio_mgmt_control_reg) = phy_reg;
88 while (*(regs->mdio_mgmt_indicators_reg) != 0)
89 ;
90
91 // 1000BASE-T Control, Address 9 (0x09)
92 *(regs->mdio_mgmt_address_reg) = 0x0009;
93 *(regs->mdio_mgmt_control_reg) = 0x1;
94 while (*(regs->mdio_mgmt_indicators_reg) != 0)
95 ;
96 *(regs->mdio_mgmt_command_reg) = 0x1;
97 while (*(regs->mdio_mgmt_indicators_reg) != 0)
98 ;
99 phy_reg = *(regs->mdio_mgmt_status_reg);
100 *(regs->mdio_mgmt_command_reg) = 0x0;
101
102 // Enable PHY is 1000BASE-T Full Duplex capable.
103 phy_reg |= 0x200; // 0b001000000000
104
105 *(regs->mdio_mgmt_address_reg) = 0x0009;
106 *(regs->mdio_mgmt_control_reg) = phy_reg;
107 while (*(regs->mdio_mgmt_indicators_reg) != 0)
108 ;
109}

111void phy_autonegotiation(eth_core_regs_t* regs)
112{
113 uint32_t phy_reg = 0xFFFF;
114 uint16_t autoneg_complete;
115 volatile uint32_t copper_aneg_timeout = 1000000u;
116 volatile uint32_t sgmii_aneg_timeout = 100000u;
117 uint8_t copper_link_up;
118
119 // Extended/GPIO Register Page Access, Address 31 (0x1F)
120 *(regs->mdio_mgmt_address_reg) = 0x001F;
121 *(regs->mdio_mgmt_control_reg) = 0x1;
122 while (*(regs->mdio_mgmt_indicators_reg) != 0)
123 ;
124
125 // Registers 0–15 are not affected by the state of the extended page
126 // register access. (see VSC8540-05 section 4.3)
127 *(regs->mdio_mgmt_address_reg) = 0x0000; // Mode Control
128 *(regs->mdio_mgmt_control_reg) = 0x1;
129 while (*(regs->mdio_mgmt_indicators_reg) != 0)
130 ;
131 *(regs->mdio_mgmt_command_reg) = 0x1;
132 while (*(regs->mdio_mgmt_indicators_reg) != 0)
133 ;
134 phy_reg = *(regs->mdio_mgmt_status_reg);
135 *(regs->mdio_mgmt_command_reg) = 0x0;
136
137 phy_reg |= 0x1200; // 0b0001001000000000
138 // Bit 9: Restart autonegotiation
139 // Bit 12: Autonegotiation enable
140 *(regs->mdio_mgmt_address_reg) = 0x0000; // Mode Control
141 *(regs->mdio_mgmt_control_reg) = phy_reg;
142 while (*(regs->mdio_mgmt_indicators_reg) != 0)
143 ;
144
145 do {
146 // Mode Status, Address 1 (0x01)
147 *(regs->mdio_mgmt_address_reg) = 0x0001;
148 *(regs->mdio_mgmt_control_reg) = 0x1;
149 while (*(regs->mdio_mgmt_indicators_reg) != 0)
150 ;
151 *(regs->mdio_mgmt_command_reg) = 0x1;
152 while (*(regs->mdio_mgmt_indicators_reg) != 0)
153 ;
154 phy_reg = *(regs->mdio_mgmt_status_reg);
155 *(regs->mdio_mgmt_command_reg) = 0x0;
156 autoneg_complete = phy_reg & 0x0020u;
157 --copper_aneg_timeout;
158 } while (!autoneg_complete && (copper_aneg_timeout != 0u));
159
160 for (volatile uint32_t i = 0; i < 100000; i++)
161 ;
162
163 *(regs->mdio_mgmt_address_reg) = 0x0001;
164 *(regs->mdio_mgmt_control_reg) = 0x1;
165 while (*(regs->mdio_mgmt_indicators_reg) != 0)
166 ;
167 *(regs->mdio_mgmt_command_reg) = 0x1;
168 while (*(regs->mdio_mgmt_indicators_reg) != 0)
169 ;
170 phy_reg = *(regs->mdio_mgmt_status_reg);
171 *(regs->mdio_mgmt_command_reg) = 0x0;
172
173 copper_link_up = phy_reg & 0x0004; // Link status
174}

176void phy_init(eth_core_regs_t* regs)
177{
178 volatile uint32_t phy_reg = 0xFFFF;
179 volatile uint32_t temp_reg = 0xFFFF;
180 volatile uint16_t id1 = 0;
181 volatile uint16_t id2 = 0;
182
183 // Select page 0
184 *(regs->mdio_mgmt_address_reg) = 0x001F;
185 *(regs->mdio_mgmt_control_reg) = 0x0000;
186 while (*(regs->mdio_mgmt_indicators_reg) != 0)
187 ;
188 *(regs->mdio_mgmt_command_reg) = 0x1;
189 while (*(regs->mdio_mgmt_indicators_reg) != 0)
190 ;
191 temp_reg = *(regs->mdio_mgmt_status_reg);
192 *(regs->mdio_mgmt_command_reg) = 0x0;
193
194 // Read PHY ID
195 *(regs->mdio_mgmt_address_reg) = 0x0002;
196 while (*(regs->mdio_mgmt_indicators_reg) != 0)
197 ;
198 *(regs->mdio_mgmt_command_reg) = 0x1;
199 while (*(regs->mdio_mgmt_indicators_reg) != 0)
200 ;
201 id1 = *(regs->mdio_mgmt_status_reg);
202 *(regs->mdio_mgmt_command_reg) = 0x0;
203
204 *(regs->mdio_mgmt_address_reg) = 0x0003;
205 while (*(regs->mdio_mgmt_indicators_reg) != 0)
206 ;
207 *(regs->mdio_mgmt_command_reg) = 0x1;
208 while (*(regs->mdio_mgmt_indicators_reg) != 0)
209 ;
210 id2 = *(regs->mdio_mgmt_status_reg);
211 *(regs->mdio_mgmt_command_reg) = 0x0;
212
213 // Select page 0
214 *(regs->mdio_mgmt_address_reg) = 0x001F;
215 *(regs->mdio_mgmt_control_reg) = 0x0000;
216 while (*(regs->mdio_mgmt_indicators_reg) != 0)
217 ;
218
219 // Extended PHY Control 1, Address 23 (0x17)
220 *(regs->mdio_mgmt_address_reg) = 0x0017;
221 *(regs->mdio_mgmt_control_reg) =
222 0x1000; // MAC interface selection to RGMII
223 while (*(regs->mdio_mgmt_indicators_reg) != 0)
224 ;
225 *(regs->mdio_mgmt_command_reg) = 0x1;
226 while (*(regs->mdio_mgmt_indicators_reg) != 0)
227 ;
228 temp_reg = *(regs->mdio_mgmt_status_reg);
229 *(regs->mdio_mgmt_command_reg) = 0x0;
230
231 // // Extended PHY Control 1, Address 24 (0x18)
232 // *(regs->mdio_mgmt_address_reg) = 0x0017;
233 // *(regs->mdio_mgmt_control_reg) = 0x10; // Enable Jumno Frame
234 // while (*(regs->mdio_mgmt_indicators_reg) != 0)
235 // ;
236 // *(regs->mdio_mgmt_command_reg) = 0x1;
237 // while (*(regs->mdio_mgmt_indicators_reg) != 0)
238 // ;
239 // temp_reg = *(regs->mdio_mgmt_status_reg);
240 // *(regs->mdio_mgmt_command_reg) = 0x0;
241
242 // Mode Control, Address 0 (0x00)
243 *(regs->mdio_mgmt_address_reg) = 0x0000;
244 *(regs->mdio_mgmt_command_reg) = 0x1;
245 while (*(regs->mdio_mgmt_indicators_reg) != 0)
246 ;
247 phy_reg = *(regs->mdio_mgmt_status_reg);
248 *(regs->mdio_mgmt_command_reg) = 0x0;
249
250 // SW reset
251 *(regs->mdio_mgmt_address_reg) = 0x0000;
252 phy_reg = phy_reg | 0x8000;
253 *(regs->mdio_mgmt_control_reg) = phy_reg;
254 while (*(regs->mdio_mgmt_indicators_reg) != 0)
255 ;
256
257 // Wait for soft reset to complete
258 do {
259 *(regs->mdio_mgmt_address_reg) = 0x0000;
260 *(regs->mdio_mgmt_command_reg) = 0x1;
261 while (*(regs->mdio_mgmt_indicators_reg) != 0)
262 ;
263 temp_reg = *(regs->mdio_mgmt_status_reg);
264 *(regs->mdio_mgmt_command_reg) = 0x0;
265 } while (0 != (temp_reg & 0x8000U));
266
267 // For RGMII mode: configure register 20E2 (to access register 20E2,
268 // register 31 must be set to 2).
269 // Set bit 11 to 0 and set RX_CLK delay and TX_CLK delay accordingly
270 // through bit [6:4] and/or bit [2:0]
271 // respectively.
272
273 // Select page 2
274 *(regs->mdio_mgmt_address_reg) = 0x001F;
275 *(regs->mdio_mgmt_control_reg) = 0x0002;
276 while (*(regs->mdio_mgmt_indicators_reg) != 0)
277 ;
278
279 // Adjust MAC interface slew rate - default is 111 but recommended for
280 // 3.3V i/o is 100. this is an extra step that is being done for the RTPF
281 // evaluation kit, as this uses 3.3V pins (bank 4 : ethernet phy pins)
282 // Adjust MAC interface slew rate
283 *(regs->mdio_mgmt_address_reg) = 0x001B;
284 *(regs->mdio_mgmt_command_reg) = 0x1;
285 while (*(regs->mdio_mgmt_indicators_reg) != 0)
286 ;
287 temp_reg = *(regs->mdio_mgmt_status_reg);
288 *(regs->mdio_mgmt_command_reg) = 0x0;
289
290 temp_reg = temp_reg & 0xFF1FU;
291 temp_reg = temp_reg | 0x0080U;
292
293 *(regs->mdio_mgmt_address_reg) = 0x001B;
294 *(regs->mdio_mgmt_control_reg) = temp_reg;
295 while (*(regs->mdio_mgmt_indicators_reg) != 0)
296 ;
297
298 // Select back page 0
299 *(regs->mdio_mgmt_address_reg) = 0x001F;
300 *(regs->mdio_mgmt_control_reg) = 0x0000;
301 while (*(regs->mdio_mgmt_indicators_reg) != 0)
302 ;
303
304 *(regs->mdio_mgmt_address_reg) = 0x0001;
305 while (*(regs->mdio_mgmt_indicators_reg) != 0)
306 ;
307 *(regs->mdio_mgmt_command_reg) = 0x1;
308 temp_reg = *(regs->mdio_mgmt_status_reg);
309 *(regs->mdio_mgmt_command_reg) = 0x0;
310}

315void tse_init(eth_core_regs_t* regs)
316{
317 uint32_t tse_reg = 0xFFFF;
318 uint32_t phy_reg = 0x0;
319 uint32_t temp = 0x0;
320
321 tse_reg = *(regs->mac_fifo_config_reg);
322

324 *(regs->mac_config_1_reg) = 0x00000005; // Tx and Rx enable
325 *(regs->mac_config_2_reg) = 0x00007213; // Full Duplex, CRC, len checking

328 *(regs->mac_max_frame_len_reg) = 0x00001000; // Maximum frame size
329 *(regs->station_address_lower_reg) =
330 0x6060603C; // Destination address lower
331 *(regs->station_address_higher_reg) =
332 0xB1C00000; // Destination address higher
333 *(regs->mac_fifo_config_reg0) = 0x0000FF00;
334 *(regs->mac_fifo_config_reg1) = 0x0FFF0000;
335 *(regs->mac_fifo_config_reg2) = 0x04000180;
336 *(regs->mac_fifo_config_reg3) = 0x0680FFFF;
337 *(regs->mac_fifo_config_reg4) = 0x00000000;
338 *(regs->mac_fifo_config_reg5) = 0x0007FFFF;
339 *(regs->frame_control_filter_reg) = 0x0000000F;
340 *(regs->mdio_mgmt_config_reg) = 0x0007; // Source clock divided by 28
341 *(regs->interface_control_reg) = 0x4; // Clear all MDIO counters
342}
