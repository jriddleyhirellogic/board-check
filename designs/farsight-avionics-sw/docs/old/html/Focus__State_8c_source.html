35State_t Focus_State( Context_t *context, Event_t *event )
36{
37 State_t next_state = (State_t)Focus_State;
38 int32_t i32_Position, i32_Setpoint, i32_Stepper_Out;
39 int32_t i32_Error;
40
41 switch( event->no_data.sig )
42 {
43 case ENTRANCE:
44 Update_Period_Ticks = pdMS_TO_TICKS( UPDATE_PERIOD_MS );
45
46 v_Set_LVDT_Gain( LVDT_GAIN_1 );
47 v_Set_Microstep_Mode( STEP_256 );
48
49 // Get current location
50 i32_Position = i32_Get_Position();
51
52 // Initialize the path planner with current position and desired final position
53 v_Init_Path_Planner( i32_Position, event->position.i32_Position );
54 i32_Final_Destination = event->position.i32_Position;
55
56 v_Start_Focus_Timer(Update_Period_Ticks);
57 v_Start_Focus_Fault_Timer( FAULT_TIMEOUT_TICKS );
58
59 Ticks_So_Far = 0;
60 DB_Count = 0; // Hysteresis count for the deadband
61 break;
62
63 case EXIT:
64 v_Set_Stepper_out( 0 );
65 v_Stop_Focus_Timer();
66 v_Stop_Focus_Fault_Timer();
67 break;
68
69 case FOCUS_TIMER_EXPIRED:
70 // Get current location
71 i32_Position = i32_Get_Position();
72
73 // Get setpoint - where it should be
74 i32_Setpoint = i32__Update_Path_SM();
75
76 i32_Error = i32_Position - i32_Final_Destination;
77
78 // Update PID controller
79 i32_Stepper_Out = i32_Update_PID( context->p_PID_Instance,
80 i32_Setpoint,
81 i32_Position,
82 0 );
83
84 v_Set_Stepper_out( i32_Stepper_Out );
85
86 //v_Set_Debug_GPIO( J7_PIN_6 , J7_PIN_6 );
87 v_Start_Focus_Timer(Update_Period_Ticks);
88
89 // Test to see how many ticks the error is in the deadband
90 if( ABS(i32_Error) <= DEADBAND_MICRONS )
91 {
92 if( ++DB_Count == DB_HYST )
93 next_state = (State_t)Idle_State;
94 }
95 else
96 {
97 DB_Count = 0;
98 }
99 break;
100
101 default:
102 break;
103 }
104
105 return next_state;
106
107}
