68void v_Update_Serial_SM(uint8_t u8_Received_Char)
69{
70 Event_t event;
71 //u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
72
73 switch( Current_State )
74 {
75 case SYNC_WAIT: // Idle state waiting for msg start
76 u8_Bytes_Received = 0;
77 if (u8_Received_Char == SYNC1_CODE )
78 {
79 Next_State = SYNC1;
80 u8_Rx_Buf[u8_Bytes_Received++] = SYNC1_CODE;
81 }
82 else
83 Next_State = SM_ERROR;
84 break;
85
86 case SYNC1: // Received SYNC1
87 if (u8_Received_Char == SYNC2_CODE )
88 {
89 Next_State = SYNC2;
90 u8_Rx_Buf[u8_Bytes_Received++] = SYNC2_CODE;
91 }
92 else
93 Next_State = SM_ERROR;
94 break;
95
96 case SYNC2: // Received SYNC2
97 if (u8_Received_Char == SYNC3_CODE )
98 {
99 Next_State = SYNC3;
100 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
101 }
102 else
103 Next_State = SM_ERROR;
104 break;
105
106 case SYNC3: // Received SYNC3
107 if (u8_Received_Char == SYNC4_CODE )
108 {
109 Next_State = SYNC4;
110 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
111 memset(&Header, 0, sizeof(Serial_Msg_Header_t));
112 }
113 else
114 Next_State = SM_ERROR;
115 break;
116
117 case SYNC4: // Received last SYNC_CHAR
118 Header.u8_Opcode = u8_Received_Char;
119 u8_Rx_Buf[u8_Bytes_Received++] = Header.u8_Opcode;
120 u32_Calculated_CRC = crc( &u8_Received_Char, 1, 0 );
121 Next_State = FUNCTION;
122 break;
123
124 case FUNCTION: // Received function code
125 Header.u8_Error_Code = u8_Received_Char;
126 u32_Calculated_CRC = crc( &u8_Received_Char, 1, u32_Calculated_CRC );
127 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
128 Next_State = ERROR_CODE;
129 break;
130
131 case ERROR_CODE:
132 Header.u8_Sequence = u8_Received_Char;
133 u32_Calculated_CRC = crc( &u8_Received_Char, 1, u32_Calculated_CRC );
134 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
135 Next_State = SEQUENCE;
136 break;
137
138 case SEQUENCE:
139 Header.u8_Data_Length = u8_Received_Char; // Size of data payload
140 u32_Calculated_CRC = crc( &u8_Received_Char, 1, u32_Calculated_CRC );
141 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
142 u8_Cnt = 0;
143 Next_State = RESERVED;
144 break;
145
146 case RESERVED:
147 Header.u8_Reserved[u8_Cnt++] = u8_Received_Char;
148 u32_Calculated_CRC = crc( &u8_Received_Char, 1, u32_Calculated_CRC );
149 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
150 if( u8_Cnt == 8 )
151 {
152 if( Header.u8_Data_Length == 0 ) // Command with no data payload
153 {
154 Next_State = CRC1;
155 }
156 else
157 {
158 Next_State = DATA;
159 }
160 }
161 break;
162
163 case DATA:
164 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
165 u32_Calculated_CRC = crc( &u8_Received_Char, 1, u32_Calculated_CRC );
166 if( u8_Bytes_Received == ( SYNC_LENGTH + HEADER_SIZE + Header.u8_Data_Length))
167 {
168 Next_State = CRC1;
169 }
170 break;
171
172 case CRC1:
173 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
174 u32_Received_CRC = u8_Received_Char;
175 Next_State = CRC2;
176 break;
177
178 case CRC2:
179 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
180 u32_Received_CRC = (u32_Received_CRC << 8) | u8_Received_Char;
181 Next_State = CRC3;
182 break;
183
184 case CRC3:
185 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
186 u32_Received_CRC = (u32_Received_CRC << 8) | u8_Received_Char;
187 Next_State = CRC4;
188 break;
189
190 case CRC4:
191 u8_Rx_Buf[u8_Bytes_Received++] = u8_Received_Char;
192 u32_Received_CRC = (u32_Received_CRC << 8) | u8_Received_Char;
193 break;
194
195 case SM_ERROR: // absorbing state until reset by timeout
196 break;
197 default:
198 break;
199 }
200
201 Current_State = Next_State;
202}

204// The timer is reloaded with every received character. The expiration is set to 3 character times. When the timer expires, the crc
205// is checked and if good, the serial comm task is notified. The state machines and crcs are reset as well.
206void v_Framing_Timeout(void)
207{
208 if( u32_Received_CRC == u32_Calculated_CRC &&
209 Current_State == CRC4 )
210 {
211 Received_Message_Notification();
212 }
213
214 Reset_Serial_SM();
215 Current_State = Next_State = SYNC_WAIT;
216}

218static void Received_Message_Notification(void)
219{

223 memcpy( &event.serial_msg.header, &Header, sizeof(Serial_Msg_Header_t));

225 v_Post_Serial_Event_ISR( &event );
226}

228void Reset_Serial_SM(void)
