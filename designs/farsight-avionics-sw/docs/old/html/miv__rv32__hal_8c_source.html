145/*------------------------------------------------------------------------------
146 * Increment value for the mtimecmp register in order to achieve a system tick
147 * interrupt as specified through the MRV_systick_config() function.
148 */
149static uint64_t g_systick_increment = 0U;
150static uint64_t g_systick_cmp_value = 0U;
151
152/*------------------------------------------------------------------------------
153 * Configure the machine timer to generate an interrupt.
154 */
155uint32_t MRV_systick_config(uint64_t ticks)
156{
157 uint32_t ret_val = ERROR;
158 uint64_t remainder = ticks;
159
160 while (remainder >= MTIME_PRESCALER)
161 {
162 remainder -= MTIME_PRESCALER;
163 g_systick_increment++;
164 }
165
166 g_systick_cmp_value = g_systick_increment + MTIME;
167
168 if (g_systick_increment > 0U)
169 {
170 WRITE_MTIMECMP(g_systick_cmp_value);
171 set_csr(mie, MIP_MTIP);
172 MRV_enable_interrupts();
173 ret_val = SUCCESS;
174 }
175
176 return ret_val;
177}

179/*------------------------------------------------------------------------------
180 * RISC-V interrupt handler for machine timer interrupts.
181 */
182void handle_m_timer_interrupt(void)
183{
184 clear_csr(mie, MIP_MTIP);
185
186 uint64_t mtime_at_irq = MTIME;
187
188#ifndef NDEBUG
189 static volatile uint32_t d_tick = 0u;
190#endif
191
192 while(g_systick_cmp_value < (mtime_at_irq + MTIME_DELTA)) {
193 g_systick_cmp_value = g_systick_cmp_value + g_systick_increment;
194
195#ifndef NDEBUG
196 d_tick += 1;
197#endif
198 }
199
200 /*
201 * Note: If d_tick > 1 it means, that a system timer interrupt has been missed.
202 *
203 * Please ensure that interrupt handlers are as short as possible to prevent
204 * them stopping other interrupts from being handled. For example, if a
205 * system timer interrupt occurs during a software interrupt, the system
206 * timer interrupt will not be handled until the software interrupt handling
207 * is complete. If the software interrupt handling time is more than one systick
208 * interval, it will result in d_tick > 1.
209 *
210 * If you are running the program using the debugger and halt the CPU at a breakpoint,
211 * MTIME will continue to increment and interrupts will be missed; resulting
212 * in d_tick > 1.
213 */
214
215 WRITE_MTIMECMP(g_systick_cmp_value);
216
217 SysTick_Handler();
218
219 set_csr(mie, MIP_MTIP);
220}

252/*------------------------------------------------------------------------------
253 * Trap handler. This function is invoked in the non-vectored mode.
254 */
255void handle_trap(uintptr_t mcause, uintptr_t mepc)
256{
257 if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == IRQ_M_SOFT))
258 {
259 handle_m_soft_interrupt();
260 }
261 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == IRQ_M_TIMER))
262 {
263 handle_m_timer_interrupt();
264 }
265 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == IRQ_M_EXT))
266 {
267#ifdef MIV_LEGACY_RV32
268 handle_m_ext_interrupt();
269#else
270 External_IRQHandler();
271#endif
272 }
273#ifndef MIV_LEGACY_RV32
274 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI0))
275 {
276 MSYS_EI0_IRQHandler();
277 }
278 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI1))
279 {
280 MSYS_EI1_IRQHandler();
281 }
282 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI2))
283 {
284 MSYS_EI2_IRQHandler();
285 }
286 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI3))
287 {
288 MSYS_EI3_IRQHandler();
289 }
290 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI4))
291 {
292 MSYS_EI4_IRQHandler();
293 }
294 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MSYS_EI5))
295 {
296 MSYS_EI5_IRQHandler();
297 }
298 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == OPSRV_REG))
299 {
300 OPSRV_IRQHandler();
301 }
302 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MGEUI))
303 {
304 MGEUI_IRQHandler();
305 }
306 else if ((mcause & MCAUSE_INT) && ((mcause & MCAUSE_CAUSE) == MGECI))
307 {
308 MGECI_IRQHandler();
309 }
310#endif /* MIV_LEGACY_RV32 */
311
312 else
313 {
314#ifndef NDEBUG
315 /*
316 Arguments supplied to this function are mcause, mepc (exception PC) and
317 stack pointer.
318 Based on privileged-isa specification mcause values and meanings are:
319
320 0 Instruction address misaligned (mtval/mtval is the address)
321 1 Instruction access fault (mtval/mtval is the address)
322 2 Illegal instruction (mtval/mtval contains the
323 offending instruction opcode)
324 3 Breakpoint
325 4 Load address misaligned (mtval/mtval is the address)
326 5 Load address fault (mtval/mtval is the address)
327 6 Store/AMO address fault (mtval/mtval is the address)
328 7 Store/AMO access fault (mtval/mtval is the address)
329 8 Environment call from U-mode
330 9 Environment call from S-mode
331 A Environment call from M-mode
332 B Instruction page fault
333 C Load page fault (mtval/mtval is the address)
334 E Store page fault (mtval/mtval is the address)
335
336 # Please note: mtval is the newer name for register mbadaddr
337 # If you get a compile failure here, use the older name.
338 # At this point, both are supported in latest compiler, older compiler
339 # versions only support mbadaddr.
340 # See: https://github.com/riscv/riscv-gcc/issues/133
341 */
342
343 /* interrupt pending */
344 uintptr_t mip = read_csr(mip);
345
346 /* additional info and meaning depends on mcause */
347 uintptr_t mtval = read_csr(mtval);
348
349 /* trap vector */
350 uintptr_t mtvec = read_csr(mtvec);
351
352 /* temporary, sometimes might hold temporary value of a0 */
353 uintptr_t mscratch = read_csr(mscratch);
354
355 /* status contains many smaller fields: */
356 uintptr_t mstatus = read_csr(mstatus);
357
358 /* PC value when the exception was taken*/
359 uintptr_t mmepc = read_csr(mepc);
360
361 /* breakpoint */
362 __asm__("ebreak");
363#else
364 _exit(1 + mcause);
365#endif /* NDEBUG */
366 }
367}
