28State_t Zero_Coarse_State( Context_t *context, Event_t *event )
29{
30 State_t next_state = (State_t)Zero_Coarse_State;
31 int32_t i32_Projection;
32 LVDT_Sample_t LVDT_Sample;
33
34 switch( event->no_data.sig )
35 {
36 case ENTRANCE:
37 v_Start_Focus_Timer(10); // Give everything a chance to settle first
38 v_Set_LVDT_Gain( LVDT_GAIN_1 );
39 v_Set_Microstep_Mode( STEP_256 );
40 break;
41
42 case EXIT:
43 v_Stop_Focus_Timer();
44 v_Set_Stepper_out( 0 );
45 break;
46
47 case FOCUS_TIMER_EXPIRED:
48 v_Start_Focus_Timer(10);
49 v_Set_Debug_GPIO( J7_PIN_6 , J7_PIN_6 );
50
51 V_Get_LVDT_Snapshot( &LVDT_Sample );
52 i32_Projection = i32_Get_Projection( &LVDT_Sample );
53
54 if( ABS( i32_Projection ) <= 30000 )
55 {
56 next_state = (State_t)Zero_Fine_State;
57 context->param = (void *)( i32_Projection < 0 ? -1 : 1 ); // remember direction of travel on entry to fine zero state
58 }
59
60 if( i32_Projection < 0 )
61 v_Set_Stepper_out( STEP_SPEED ); // Move in negative direction
62 else
63 v_Set_Stepper_out( -STEP_SPEED ); // Move in positive direction
64
65 break;
66
67 default:
68 break;
69 }
70
71 return next_state;
72
73}
