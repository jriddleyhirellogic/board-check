433static inline int test_and_set_bit(int nr, volatile unsigned long *addr)
434{
435 unsigned long __res, __mask;
436 __mask = MASK(nr);
437 __asm__ __volatile__ ( \
438 "amoor.w %0, %2, %1" \
439 : "=r" (__res), "+A" (*addr) \
440 : "r" (__mask)); \
441
442 return ((__res & __mask) != 0);
443}
