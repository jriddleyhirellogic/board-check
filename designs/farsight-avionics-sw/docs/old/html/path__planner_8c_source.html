77// This supplies a target setpoint path to the PI controller to follow with each time tick
78int32_t i32__Update_Path_SM(void)
79{
80 int32_t i32_V_Temp;
81 float f_Interval_Distance_Travelled;
82 int32_t i32_Setpoint; // Output to PI controller
83 int32_t i32_Acceleration = 0;
84
85 // Direction from the start displacement to end
86 int32_t i32_Direction = (i32_Final_Position_microns - i32_Start_Position_microns) >= 0 ? 1 : -1;
87
88 f_Elapsed_Time_sec += T;
89
90 switch(current_state)
91 {
92 case INIT:
93 next_state = ACCELERATE;
94 i32_Current_Position_microns = i32_Start_Position_microns;
95 f_Current_Distance_Travelled_microns = 0.0f;
96 f_Total_Distance_microns = (float)ABS((i32_Final_Position_microns - i32_Start_Position_microns));
97 i32_Current_Velocity = 0;
98 break;
99
100 case ACCELERATE: // Accelerating - hasn't reached halfway point yet
101 i32_Acceleration = i32_Direction * Motion.a;
102
103 // See if velocity after acceleration will exceed max or min
104 i32_V_Temp = i32_Current_Velocity + i32_Acceleration;
105 if( i32_V_Temp > Motion.v_max )
106 {
107 i32_V_Temp = Motion.v_max;
108 f_Ramp_Distance_microns = f_Current_Distance_Travelled_microns;
109 i32_Acceleration = 0;
110 next_state = CONSTANT_V;
111 break;
112 }
113 else if( i32_V_Temp < Motion.v_min )
114 {
115 i32_V_Temp = Motion.v_min;
116 f_Ramp_Distance_microns = f_Current_Distance_Travelled_microns;
117 i32_Acceleration = 0;
118 next_state = CONSTANT_V;
119 break;
120 }
121
122 // If gone at least halfway, then triangular profile, start decelerating
123 if( f_Current_Distance_Travelled_microns >= f_Total_Distance_microns/2.0f )
124 {
125 next_state = DECELERATE;
126 }
127 break;
128
129 case CONSTANT_V: // Constant Velocity - velocity has hit max, but not halfway yet
130 i32_Acceleration = 0;
131
132 if( f_Distance_Remaining_microns <= f_Ramp_Distance_microns)
133 {
134 next_state = DECELERATE;
135 }
136 break;
137
138 case DECELERATE: // Decelerating
139 i32_Acceleration = -i32_Direction * Motion.a;
140
141 if( ABS(f_Distance_Remaining_microns) <= f_Tolerance_microns )
142 {
143 next_state = DONE;
144 }
145 break;
146
147 case DONE:
148 // Assumption is that PI controller will drive it the rest of the way and only uses the setpoint
149 break;
150
151 default:
152 break;
153 }
154
155 // Update velocity and bound
156 i32_Current_Velocity = i32_Current_Velocity + i32_Acceleration;
157 i32_Current_Velocity = i32_Current_Velocity < Motion.v_min ? Motion.v_min : i32_Current_Velocity;
158 i32_Current_Velocity = i32_Current_Velocity > Motion.v_max ? Motion.v_max : i32_Current_Velocity;
159
160 f_Interval_Distance_Travelled = fabs( v_base * T * ( (float)(i32_Current_Velocity + (float)i32_Acceleration/2.0f )) );
161
162 assert( !isnan(f_Interval_Distance_Travelled ));
163 assert( !isnan( f_Current_Distance_Travelled_microns ));
164
165 f_Current_Distance_Travelled_microns += f_Interval_Distance_Travelled;
166 f_Distance_Remaining_microns = fabs( f_Total_Distance_microns - f_Current_Distance_Travelled_microns );
167
168
169 i32_Setpoint = current_state != DONE ? i32_Start_Position_microns + (int32_t)f_Current_Distance_Travelled_microns * i32_Direction :
170 i32_Final_Position_microns;
171
172 current_state = next_state;
173
174 return i32_Setpoint;
175}
