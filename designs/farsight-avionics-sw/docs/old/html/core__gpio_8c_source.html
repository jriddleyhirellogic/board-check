31void GPIO_init
32(
33 gpio_instance_t * this_gpio,
34 addr_t base_addr,
35 gpio_apb_width_t bus_width
36)
37{
38 uint8_t i = 0;
39 addr_t cfg_reg_addr = base_addr;
40
41 this_gpio->base_addr = base_addr;
42 this_gpio->apb_bus_width = bus_width;
43
44 /* Clear configuration. */
45 for( i = 0, cfg_reg_addr = base_addr; i < NB_OF_GPIO; ++i )
46 {
47 HW_set_8bit_reg( cfg_reg_addr, 0 );
48 cfg_reg_addr += 4;
49 }
50 /* Clear any pending interrupts */
51 switch( this_gpio->apb_bus_width )
52 {
53 case GPIO_APB_32_BITS_BUS:
54 HAL_set_32bit_reg( this_gpio->base_addr, IRQ, CLEAR_ALL_IRQ32 );
55 break;
56
57 case GPIO_APB_16_BITS_BUS:
58 HAL_set_16bit_reg( this_gpio->base_addr, IRQ0, (uint16_t)CLEAR_ALL_IRQ16 );
59 HAL_set_16bit_reg( this_gpio->base_addr, IRQ1, (uint16_t)CLEAR_ALL_IRQ16 );
60 break;
61
62 case GPIO_APB_8_BITS_BUS:
63 HAL_set_8bit_reg( this_gpio->base_addr, IRQ0, (uint8_t)CLEAR_ALL_IRQ8 );
64 HAL_set_8bit_reg( this_gpio->base_addr, IRQ1, (uint8_t)CLEAR_ALL_IRQ8 );
65 HAL_set_8bit_reg( this_gpio->base_addr, IRQ2, (uint8_t)CLEAR_ALL_IRQ8 );
66 HAL_set_8bit_reg( this_gpio->base_addr, IRQ3, (uint8_t)CLEAR_ALL_IRQ8 );
67 break;
68
69 default:
70 HAL_ASSERT(0);
71 break;
72 }
73}

79void GPIO_config
80(
81 gpio_instance_t * this_gpio,
82 gpio_id_t port_id,
83 uint32_t config
84)
85{
86 HAL_ASSERT( port_id < NB_OF_GPIO );
87
88 if ( port_id < NB_OF_GPIO )
89 {
90 uint32_t cfg_reg_addr = this_gpio->base_addr;
91 cfg_reg_addr += (port_id * 4);
92 HW_set_32bit_reg( cfg_reg_addr, config );
93
94 /*
95 * Verify that the configuration was correctly written. Failure to read
96 * back the expected value may indicate that the GPIO port was configured
97 * as part of the hardware flow and cannot be modified through software.
98 * It may also indicate that the base address passed as parameter to
99 * GPIO_init() was incorrect.
100 */
101 HAL_ASSERT( HW_get_32bit_reg( cfg_reg_addr ) == config );
102 }

109void GPIO_set_outputs
110(
111 gpio_instance_t * this_gpio,
112 uint32_t value
113)
114{
115 uint32_t readback;
116
117 switch( this_gpio->apb_bus_width )
118 {
119 case GPIO_APB_32_BITS_BUS:
120 HAL_set_32bit_reg( this_gpio->base_addr, GPIO_OUT, value );
121 break;
122
123 case GPIO_APB_16_BITS_BUS:
124 HAL_set_16bit_reg( this_gpio->base_addr, GPIO_OUT0, (uint16_t)value );
125 HAL_set_16bit_reg( this_gpio->base_addr, GPIO_OUT1, (uint16_t)(value >> 16) );
126 break;
127
128 case GPIO_APB_8_BITS_BUS:
129 HAL_set_8bit_reg( this_gpio->base_addr, GPIO_OUT0, (uint8_t)value );
130 HAL_set_8bit_reg( this_gpio->base_addr, GPIO_OUT1, (uint8_t)(value >> 8) );
131 HAL_set_8bit_reg( this_gpio->base_addr, GPIO_OUT2, (uint8_t)(value >> 16) );
132 HAL_set_8bit_reg( this_gpio->base_addr, GPIO_OUT3, (uint8_t)(value >> 24) );
133 break;
134
135 default:
136 HAL_ASSERT(0);
137 break;
138 }
139
140 /*
141 * Verify that the output register was correctly written. Failure to read back
142 * the expected value may indicate that some of the GPIOs may not exist due to
143 * the number of GPIOs selected in the CoreGPIO hardware flow configuration.
144 * It may also indicate that the base address or APB bus width passed as
145 * parameter to the GPIO_init() function do not match the hardware design.
146 */
147 readback = GPIO_get_outputs( this_gpio );
148
149 HAL_ASSERT( readback == value );
150}

230void GPIO_set_output
231(
232 gpio_instance_t * this_gpio,
233 gpio_id_t port_id,
234 uint8_t value
235)
236{
237 HAL_ASSERT( port_id < NB_OF_GPIO );
238
239
240 switch( this_gpio->apb_bus_width )
241 {
242 case GPIO_APB_32_BITS_BUS:
243 {
244 uint32_t outputs_state;
245
246 outputs_state = HAL_get_32bit_reg( this_gpio->base_addr, GPIO_OUT );
247 if ( 0 == value )
248 {
249 outputs_state &= ~(1 << port_id);
250 }
251 else
252 {
253 outputs_state |= 1 << port_id;
254 }
255 HAL_set_32bit_reg( this_gpio->base_addr, GPIO_OUT, outputs_state );
256
257 /*
258 * Verify that the output register was correctly written. Failure to read back
259 * the expected value may indicate that some of the GPIOs may not exist due to
260 * the number of GPIOs selected in the CoreGPIO hardware flow configuration.
261 * It may also indicate that the base address or APB bus width passed as
262 * parameter to the GPIO_init() function do not match the hardware design.
263 */
264 HAL_ASSERT( HAL_get_32bit_reg( this_gpio->base_addr, GPIO_OUT ) == outputs_state );
265 }
266 break;
267
268 case GPIO_APB_16_BITS_BUS:
269 {
270 uint16_t outputs_state;
271 uint32_t gpio_out_reg_addr = this_gpio->base_addr + GPIO_OUT_REG_OFFSET + ((port_id >> 4) * 4);
272
273 outputs_state = HW_get_16bit_reg( gpio_out_reg_addr );
274 if ( 0 == value )
275 {
276 outputs_state &= ~(1 << (port_id & 0x0F));
277 }
278 else
279 {
280 outputs_state |= 1 << (port_id & 0x0F);
281 }
282 HW_set_16bit_reg( gpio_out_reg_addr, outputs_state );
283
284 /*
285 * Verify that the output register was correctly written. Failure to read back
286 * the expected value may indicate that some of the GPIOs may not exist due to
287 * the number of GPIOs selected in the CoreGPIO hardware flow configuration.
288 * It may also indicate that the base address or APB bus width passed as
289 * parameter to the GPIO_init() function do not match the hardware design.
290 */
291 HAL_ASSERT( HW_get_16bit_reg( gpio_out_reg_addr ) == outputs_state );
292 }
293 break;
294
295 case GPIO_APB_8_BITS_BUS:
296 {
297 uint8_t outputs_state;
298 uint32_t gpio_out_reg_addr = this_gpio->base_addr + GPIO_OUT_REG_OFFSET + ((port_id >> 3) * 4);
299
300 outputs_state = HW_get_8bit_reg( gpio_out_reg_addr );
301 if ( 0 == value )
302 {
303 outputs_state &= ~(1 << (port_id & 0x07));
304 }
305 else
306 {
307 outputs_state |= 1 << (port_id & 0x07);
308 }
309 HW_set_8bit_reg( gpio_out_reg_addr, outputs_state );
310
311 /*
312 * Verify that the output register was correctly written. Failure to read back
313 * the expected value may indicate that some of the GPIOs may not exist due to
314 * the number of GPIOs selected in the CoreGPIO hardware flow configuration.
315 * It may also indicate that the base address or APB bus width passed as
316 * parameter to the GPIO_init() function do not match the hardware design.
317 */
318 HAL_ASSERT( HW_get_8bit_reg( gpio_out_reg_addr ) == outputs_state );
319 }
320 break;
321
322 default:
323 HAL_ASSERT(0);
324 break;
325 }
326}

332void GPIO_drive_inout
333(
334 gpio_instance_t * this_gpio,
335 gpio_id_t port_id,
336 gpio_inout_state_t inout_state
337)
338{
339 uint32_t config;
340 uint32_t cfg_reg_addr = this_gpio->base_addr;
341
342 HAL_ASSERT( port_id < NB_OF_GPIO );
343
344 switch( inout_state )
345 {
346 case GPIO_DRIVE_HIGH:
347 /* Set output high */
348 GPIO_set_output( this_gpio, port_id, 1 );
349
350 /* Enable output buffer */
351 cfg_reg_addr = this_gpio->base_addr + (port_id * 4);
352 config = HW_get_8bit_reg( cfg_reg_addr );
353 config |= OUTPUT_BUFFER_ENABLE_MASK;
354 HW_set_8bit_reg( cfg_reg_addr, config );
355 break;
356
357 case GPIO_DRIVE_LOW:
358 /* Set output low */
359 GPIO_set_output( this_gpio, port_id, 0 );
360
361 /* Enable output buffer */
362 cfg_reg_addr = this_gpio->base_addr + (port_id * 4);
363 config = HW_get_8bit_reg( cfg_reg_addr );
364 config |= OUTPUT_BUFFER_ENABLE_MASK;
365 HW_set_8bit_reg( cfg_reg_addr, config );
366 break;
367
368 case GPIO_HIGH_Z:
369 /* Disable output buffer */
370 cfg_reg_addr = this_gpio->base_addr + (port_id * 4);
371 config = HW_get_8bit_reg( cfg_reg_addr );
372 config &= ~OUTPUT_BUFFER_ENABLE_MASK;
373 HW_set_8bit_reg( cfg_reg_addr, config );
374 break;
375
376 default:
377 HAL_ASSERT(0);
378 break;
379 }
380}
